processar_voos <- function(chunk, pos) {
  chunk |>
    filter(!startsWith(DESTINATION_AIRPORT, "1")) |>
    drop_na() |>
    group_by(DESTINATION_AIRPORT) |>
    summarise(
      soma_atraso = sum(ARRIVAL_DELAY),
      qtd_voos = n(),
      .groups = 'drop'
    )
}

estatisticas_chunks <- read_csv_chunked(
  "flights.csv",
  callback = DataFrameCallback$new(processar_voos),
  chunk_size = 1000000,
  col_types = cols_only(
    DESTINATION_AIRPORT = col_character(),
    ARRIVAL_DELAY = col_double()
  )
)

flights <- estatisticas_chunks |>
  group_by(DESTINATION_AIRPORT) |>
  summarise(
    soma_total = sum(soma_atraso),
    voos_totais = sum(qtd_voos),
    .groups = 'drop'
  ) |>
  mutate(ATRASO_MEDIO = soma_total / voos_totais) |>
  select(DESTINATION_AIRPORT, ATRASO_MEDIO)
